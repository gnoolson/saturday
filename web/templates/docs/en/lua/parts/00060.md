### Conditions

```
local flag = true

if flag then
-- action
end

flag = false

if not flag then
-- action
end

local x = 60

if x == 100 then
-- action
elseif x ~= 60 then
-- "~=" means "not equal"
elseif x < 18 then
-- action
elseif (x >= 16) and (x < 18) then
-- logical "and"
elseif (x > 10000) or (x < -10000) then
-- logical "or"
else
-- executed if none of the conditions are met
end
```
